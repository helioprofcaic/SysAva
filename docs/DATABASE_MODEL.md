# 🗄️ Modelo do Banco de Dados (SQL)

Este arquivo contém todos os comandos SQL necessários para criar a estrutura completa do banco de dados do SysAva no Supabase.

Copie e cole todo o bloco de código abaixo no **SQL Editor** do seu projeto Supabase e execute-o uma única vez.

```sql
-- =============================================================================
-- 🛠️ SETUP DO BANCO DE DADOS (SUPABASE / POSTGRESQL)
-- =============================================================================
-- Este script cria toda a estrutura de tabelas para o SysAva.
-- É seguro executá-lo múltiplas vezes, pois ele usa "CREATE TABLE IF NOT EXISTS".

-- 1. Tabela de Usuários
CREATE TABLE IF NOT EXISTS public.app_users (
    username TEXT PRIMARY KEY,
    password TEXT NOT NULL,
    name TEXT NOT NULL,
    ra TEXT UNIQUE,
    role TEXT DEFAULT 'student'
);

-- 2. Estrutura Acadêmica (Escolas, Turmas, Disciplinas)
CREATE TABLE IF NOT EXISTS public.schools (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name TEXT NOT NULL,
    gre TEXT
);

CREATE TABLE IF NOT EXISTS public.classes (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name TEXT NOT NULL,
    code TEXT UNIQUE NOT NULL,
    school_id BIGINT REFERENCES public.schools(id)
);

CREATE TABLE IF NOT EXISTS public.subjects (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name TEXT UNIQUE NOT NULL,
    type TEXT NOT NULL DEFAULT 'regular', -- 'regular' ou 'training'
    duration_type TEXT DEFAULT 'anual' -- 'anual' ou 'mensal'
);

-- Tabela de junção: Turmas <-> Disciplinas
CREATE TABLE IF NOT EXISTS public.class_subjects (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    class_id BIGINT REFERENCES public.classes(id) ON DELETE CASCADE,
    subject_id BIGINT REFERENCES public.subjects(id) ON DELETE CASCADE,
    is_active BOOLEAN DEFAULT true,
    UNIQUE (class_id, subject_id)
);

-- Matrículas dos alunos
CREATE TABLE IF NOT EXISTS public.student_enrollments (
    user_username TEXT REFERENCES public.app_users(username) ON DELETE CASCADE,
    class_id BIGINT REFERENCES public.classes(id) ON DELETE CASCADE,
    PRIMARY KEY (user_username, class_id)
);

-- 3. Conteúdo de Ensino (Aulas, Fórum, Quiz)
CREATE TABLE IF NOT EXISTS public.lessons (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    title TEXT NOT NULL,
    description TEXT,
    video_url TEXT,
    subject_id BIGINT REFERENCES public.subjects(id),
    -- Garante que não existam duas aulas com mesmo título na mesma disciplina
    CONSTRAINT lessons_subject_id_title_key UNIQUE (subject_id, title)
);

CREATE TABLE IF NOT EXISTS public.forum_posts (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    user_name TEXT NOT NULL,
    message TEXT NOT NULL,
    lesson_id BIGINT REFERENCES public.lessons(id) ON DELETE CASCADE,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS public.quizzes (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    lesson_id BIGINT REFERENCES public.lessons(id) ON DELETE CASCADE,
    title TEXT NOT NULL
);

CREATE TABLE IF NOT EXISTS public.quiz_questions (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    quiz_id BIGINT REFERENCES public.quizzes(id) ON DELETE CASCADE,
    question_text TEXT NOT NULL,
    options JSONB NOT NULL, -- Ex: ["Opção A", "Opção B", "Opção C"]
    correct_option_index INTEGER NOT NULL
);

CREATE TABLE IF NOT EXISTS public.user_history (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    username TEXT NOT NULL REFERENCES public.app_users(username) ON DELETE CASCADE,
    activity TEXT NOT NULL,
    timestamp TIMESTAMPTZ DEFAULT NOW()
);

create table public.assessments (
  id bigint generated by default as identity primary key,
  subject_id bigint references public.subjects(id),
  type text not null, -- MN1, MN2, MN3, RM
  title text,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

create table public.assessment_questions (
  id bigint generated by default as identity primary key,
  assessment_id bigint references public.assessments(id) on delete cascade,
  question_text text not null,
  question_type text default 'objective', -- 'objective' or 'subjective'
  options jsonb, -- Array de strings para objetivas
  correct_option_index int default 0,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

create table public.student_assessments (
  id bigint generated by default as identity primary key,
  user_username text references public.app_users(username),
  assessment_id bigint references public.assessments(id),
  score numeric,
  status text default 'submitted',
  submitted_at timestamp with time zone default timezone('utc'::text, now()) not null
);

create table public.student_assessment_answers (
  id bigint generated by default as identity primary key,
  submission_id bigint references public.student_assessments(id),
  question_id bigint references public.assessment_questions(id),
  selected_option_index int,
  answer_text text,
  answer_link text
);

create table public.schedules (
  id bigint generated by default as identity primary key,
  subject_id bigint references public.subjects(id) on delete cascade,
  content text,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);


CREATE TABLE attendance (
    id BIGSERIAL PRIMARY KEY,
    student_name TEXT NOT NULL,
    student_number INT,
    is_present BOOLEAN DEFAULT TRUE,
    class_name TEXT NOT NULL,
    subject_id BIGINT REFERENCES public.subjects(id),
    date DATE NOT NULL,
    professor_name TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE (student_name, class_name, subject_id, date) -- Precisão por disciplina
);

CREATE TABLE weekly_schedule (
    id BIGSERIAL PRIMARY KEY,    
    class_name TEXT NOT NULL,
    day_of_week TEXT NOT NULL,
    time_slot TEXT NOT NULL,
    subject_name TEXT NOT NULL,
    professor_name TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    class_id BIGINT, -- Armazena o código numérico da turma (convertido de classes.code)
    UNIQUE (class_name, day_of_week, time_slot) -- Impede conflitos de horário na mesma turma
);

-- =============================================================================
-- 🧹 COMANDOS ÚTEIS (MANUTENÇÃO)
-- =============================================================================
-- Limpar apenas as aulas (útil se der erro de duplicidade na importação de aulas)
-- TRUNCATE TABLE public.lessons RESTART IDENTITY CASCADE;
```
